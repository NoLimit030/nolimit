function UI:AddCleanup(callback)
    assert(type(callback) == "function", "Cleanup must be a function.")
    if destroyed then pcall(callback) else extensionCleanups[#extensionCleanups + 1] = callback end

local function connect(signal, callback)
    local c = signal:Connect(callback)
    connections[#connections + 1] = c
    return c
end

local function create(class, props, parent)
    local obj = Instance.new(class)
    for key, value in pairs(props or {}) do obj[key] = value end
    obj.Parent = parent
    return obj
end

local function round(obj, radius)
    create("UICorner", {CornerRadius = UDim.new(0, radius or 10)}, obj)
end

local function border(obj, color)
    return create("UIStroke", {Color = color or colors.line, Thickness = 1,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Transparency = 0.25}, obj)
end

local function bind(obj, prop, token)
    bindings[#bindings + 1] = {obj, prop, token}
    obj[prop] = themes[config.Theme][token]
end

local function tween(obj, props, duration)
    if activeTweens[obj] then activeTweens[obj]:Cancel(); activeTweens[obj] = nil end
    if not config.Animations then
        for key, value in pairs(props) do obj[key] = value end
        return nil
    end
    local tw = TweenService:Create(obj, TweenInfo.new(duration or 0.16,
        Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    activeTweens[obj] = tw
    tw:Play()
    return tw
end

        TextXAlignment = Enum.TextXAlignment.Left,
        TextYAlignment = Enum.TextYAlignment.Center, TextTruncate = Enum.TextTruncate.AtEnd}, parent)
end

local function button(parent, text, x, y, w, h, callback, accent)
    local b = create("TextButton", {Text = text, AutoButtonColor = false,
        Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(w, h),
        BackgroundColor3 = colors.raised, TextColor3 = colors.text,
        Font = Enum.Font.GothamMedium, TextSize = 12, BorderSizePixel = 0}, parent)
    round(b, 8)
    create("UIGradient", {Color = ColorSequence.new(Color3.fromRGB(255, 255, 255), Color3.fromRGB(221, 230, 241)), Rotation = 90}, b)
    if accent then
        bind(b, "BackgroundColor3", "accent"); b.TextColor3 = colors.bg
    else
        border(b)
        connect(b.MouseEnter, function() tween(b, {BackgroundColor3 = colors.hover}) end)
        connect(b.MouseLeave, function() tween(b, {BackgroundColor3 = colors.raised}) end)
    end
    if callback then connect(b.Activated, function() if not destroyed then callback() end end) end
    return b
end

local function padding(parent, n)
    create("UIPadding", {PaddingTop = UDim.new(0, n), PaddingBottom = UDim.new(0, n),
        PaddingLeft = UDim.new(0, n), PaddingRight = UDim.new(0, n)}, parent)
end

local function gradient(parent, first, last, rotation)
    -- UIGradient multiplies the host color; use white for literal palette colors.
    parent.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    return create("UIGradient", {Color = ColorSequence.new(first, last), Rotation = rotation or 0}, parent)
end

local function accentGradient(parent, rotation)
    local g = gradient(parent, themes[config.Theme].tint, colors.surface, rotation or 25)
    themeRefreshers[#themeRefreshers + 1] = function()
        g.Color = ColorSequence.new(themes[config.Theme].tint, colors.surface)
    end
    return g
end

function UI:SetTooltip(target, text)
    if destroyed or not target then return end
    local help = tostring(text or "")
    if help == "" then return end
    if tooltipHelp[target] then tooltipHelp[target] = help; return end
    tooltipHelp[target] = help
    connect(target.MouseEnter, function()
        if destroyed or not tooltipHost then return end
        tooltipTarget = target; tooltipText.Text = tooltipHelp[target] or help
        local relative = target.AbsolutePosition - root.AbsolutePosition
        local size = root.AbsoluteSize
        local x = math.clamp(relative.X, 8, math.max(8, size.X - 328))
        local y = relative.Y + target.AbsoluteSize.Y + 8
        if y + 88 > size.Y then y = math.max(8, relative.Y - 96) end
        tooltipHost.Position = UDim2.fromOffset(x, y); tooltipHost.Visible = true
    end)
    connect(target.MouseLeave, function()
        if tooltipHost and tooltipTarget == target then tooltipTarget = nil; tooltipHost.Visible = false end
    end)
    connect(target.Destroying, function()
        tooltipHelp[target] = nil
        if tooltipHost and tooltipTarget == target then tooltipTarget = nil; tooltipHost.Visible = false end
    end)
end
UI.AttachTooltip = UI.SetTooltip

-- A lightweight native crown; no downloads or bitmap dependencies.
local function crown(parent, x, y, size)
    local holder = create("Frame", {BackgroundTransparency = 1,
        Position = UDim2.fromOffset(x, y), Size = UDim2.fromOffset(size, size)}, parent)
    local function line(ax, ay, bx, by)
        local dx, dy = bx - ax, by - ay
        local f = create("Frame", {BorderSizePixel = 0, AnchorPoint = Vector2.new(0.5, 0.5),
            Position = UDim2.fromOffset((ax + bx) * size / 2, (ay + by) * size / 2),
            Size = UDim2.fromOffset(math.sqrt(dx * dx + dy * dy) * size, math.max(1.5, size * 0.045)),
            Rotation = math.deg(math.atan2(dy, dx))}, holder)
        bind(f, "BackgroundColor3", "accent"); round(f, 2)
    end
    line(.16, .34, .32, .53); line(.32, .53, .50, .22)
    line(.50, .22, .68, .53); line(.68, .53, .84, .34)
    line(.84, .34, .75, .75); line(.75, .75, .25, .75)
    line(.25, .75, .16, .34); line(.28, .85, .72, .85)
    return holder
end

function UI:Notify(message)
    if destroyed then return end
    toastToken = toastToken + 1
    local token = toastToken
    if toastTask then pcall(task.cancel, toastTask); toastTask = nil end
    toastText.Text = tostring(message); toast.Visible = true
    toastTask = task.delay(3.5, function()
        if not destroyed and token == toastToken then toast.Visible = false end
        toastTask = nil
    end)
end

local function refreshTheme()
    local theme = themes[config.Theme]
    for _, item in ipairs(bindings) do
        if item[1].Parent then
            if activeTweens[item[1]] then activeTweens[item[1]]:Cancel(); activeTweens[item[1]] = nil end
            item[1][item[2]] = theme[item[3]]
        end
    end
    for _, fn in ipairs(themeRefreshers) do fn() end
end

function UI:SetTheme(name)
    if destroyed or not themes[name] then return false end
    config.Theme = name; refreshTheme(); return true
end

local function viewport() return root.AbsoluteSize end
local function clampFrame(frame, width, height)
    local area = viewport()
    local x, y = frame.Position.X.Offset, frame.Position.Y.Offset
    x = math.clamp(x, 8, math.max(8, area.X - width - 8))
    y = math.clamp(y, 8, math.max(8, area.Y - height - 8))
    frame.Position = UDim2.fromOffset(x, y)
end

local function fitWindow(center)
    local area = viewport()
    if area.X < 1 or area.Y < 1 then return end
    fitScale = math.min(1, (area.X - 24) / 960, (area.Y - 24) / 640)
    uiScale = math.max(.1, fitScale * config.Scale)
    scale.Scale = uiScale
    if center then
        main.Position = UDim2.fromOffset(math.floor((area.X - 960 * uiScale) / 2), math.floor((area.Y - 640 * uiScale) / 2))
    else
        clampFrame(main, 960 * uiScale, 640 * uiScale)
    end
    -- Resolve Scale-based positions before clamping floating windows.
    metrics.Position = UDim2.fromOffset(metrics.AbsolutePosition.X - root.AbsolutePosition.X,
        metrics.AbsolutePosition.Y - root.AbsolutePosition.Y)
    clampFrame(metrics, 240, 112); clampFrame(launcher, 52, 52)
end

local function cancelOpenTween()
    if openDone then openDone:Disconnect(); openDone = nil end
    local tracked = activeTweens[scale]
    if tracked then tracked:Cancel() end
    if openTween and openTween ~= tracked then openTween:Cancel() end
    openTween = nil
    activeTweens[scale] = nil
end
function UI:SetVisible(value)
    if destroyed then return end
    visible = value == true; capturingKey = false
    if tooltipHost then tooltipHost.Visible = false; tooltipTarget = nil end
    if refreshKey then refreshKey() end
    cancelOpenTween()
    launcher.Visible = not visible
    if visible then
        main.Visible = true
        scale.Scale = config.Animations and uiScale * .975 or uiScale
        tween(scale, {Scale = uiScale}, .18)
    elseif config.Animations then
        openTween = tween(scale, {Scale = uiScale * .975}, .12)
        openDone = openTween.Completed:Connect(function(state)
            if state == Enum.PlaybackState.Completed and not visible and not destroyed then main.Visible = false end
        end)
    else main.Visible = false end
end

-- All dragging uses physical screen pixels. The content alone is scaled.
local function beginDrag(target, input, click)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
    if target == main then cancelOpenTween(); scale.Scale = uiScale end
    local absolute = target.AbsolutePosition - root.AbsolutePosition
    target.Position = UDim2.fromOffset(absolute.X, absolute.Y)
    drag = {target = target, input = input, start = input.Position,
        position = absolute, moved = false, click = click}
end
connect(dragHandle.InputBegan, function(input) beginDrag(main, input) end)
connect(metricsHandle.InputBegan, function(input) beginDrag(metrics, input) end)
connect(launcher.InputBegan, function(input) beginDrag(launcher, input, function() UI:SetVisible(true) end) end)
connect(UIS.InputChanged, function(input)
    if not drag then return end
    local touch = drag.input.UserInputType == Enum.UserInputType.Touch
    if (touch and input ~= drag.input) or (not touch and input.UserInputType ~= Enum.UserInputType.MouseMovement) then return end
    local delta = input.Position - drag.start
    if delta.Magnitude > 5 then drag.moved = true end
    if drag.moved then
        local target = drag.target
        target.Position = UDim2.fromOffset(drag.position.X + delta.X, drag.position.Y + delta.Y)
        local factor = target == main and uiScale or 1
        clampFrame(target, target.Size.X.Offset * factor, target.Size.Y.Offset * factor)
    end
end)
connect(UIS.InputEnded, function(input)
    if not drag then return end
    if input == drag.input or (drag.input.UserInputType == Enum.UserInputType.MouseButton1 and input.UserInputType == Enum.UserInputType.MouseButton1) then
        local finished = drag; drag = nil
        if not finished.moved and finished.click then finished.click() end
    end
end)
connect(UIS.WindowFocusReleased, function() drag = nil end)

local Planner = (function()
-- Pure decision rules. No Roblox services or network calls.
local Planner = {}
function Planner.isFinite(n)
    return type(n) == "number" and n == n and n > -math.huge and n < math.huge
end
function Planner.nextUpgrade(snapshot, configs, order, enabled, reserve)
    if type(snapshot) ~= "table" or type(configs) ~= "table" or type(snapshot.OwnedFoods) ~= "table" then return nil end
    local money = tonumber(snapshot.Money) or 0
    if not Planner.isFinite(money) or not Planner.isFinite(reserve or 0) then return nil end
    local available = math.max(0, money - math.max(0, reserve or 0))
    for _, kind in ipairs(order) do
        if enabled[kind] then
            local nextIndex
            if kind == "Food" then
                local highest = 0
                for i = 1, configs.Food.Count() do
                    if snapshot.OwnedFoods[tostring(i)] == true or snapshot.OwnedFoods[i] == true then highest = i end
                end
                nextIndex = highest + 1
            elseif kind == "Stand" then
                if not Planner.isFinite(snapshot.Stand) or snapshot.Stand < 0 or snapshot.Stand % 1 ~= 0 then continue end
                nextIndex = snapshot.Stand + 1
            elseif kind == "Fence" then
                if not Planner.isFinite(snapshot.FenceLevel) or snapshot.FenceLevel < 0 or snapshot.FenceLevel % 1 ~= 0 then continue end
                nextIndex = snapshot.FenceLevel + 1
            else continue end
            local item = configs[kind].Get(nextIndex)
            if item and Planner.isFinite(item.Price) and item.Price >= 0 and item.Price <= available then
                return {Kind = kind, Index = nextIndex, Price = item.Price, Item = item}
            end
        end
    end
    return nil
end
function Planner.sellCandidates(animals, capacity, incomeOf, protectedNames)
    -- Never sell before every plot slot is occupied. Compare with its weakest animal.
    if type(animals) ~= "table" or not Planner.isFinite(capacity) or capacity < 1 or capacity % 1 ~= 0
        or type(incomeOf) ~= "function" or type(protectedNames) ~= "table" then return {} end
    local placed, weakest = 0, math.huge
    for _, entry in pairs(animals) do
        if type(entry) ~= "table" then return {} end
        if entry.Placed == true then
            local income = incomeOf(entry)
            if not Planner.isFinite(income) or income < 0 then return {} end
            placed = placed + 1; weakest = math.min(weakest, income)
        end
    end
    if placed < capacity then return {} end
    local result = {}
    for id, entry in pairs(animals) do
        local income = incomeOf(entry)
        local name = tostring(entry.ModelName or ""):lower()
        if type(entry.ModelName) == "string" and entry.ModelName ~= "" and entry.Placed == false and not entry.Bought and not entry.Favorite and not entry.Favorited
            and not entry.Locked and not entry.Protected and not protectedNames[name]
            and Planner.isFinite(income) and income >= 0 and income < weakest then
            result[#result + 1] = {Id = id, Income = income, Entry = entry}
        end
    end
    table.sort(result, function(a, b)
        if a.Income == b.Income then return tostring(a.Id) < tostring(b.Id) end
        return a.Income < b.Income
    end)
    return result
end
function Planner.previewAnimals(animals, capacity, incomeOf, protectedNames)
    if type(animals) ~= "table" then return {} end
    local eligible = {}
    for _, candidate in ipairs(Planner.sellCandidates(animals, capacity, incomeOf, protectedNames)) do eligible[candidate.Id] = true end
    local placed, weakest = 0, math.huge
    local validCapacity = Planner.isFinite(capacity) and capacity >= 1 and capacity % 1 == 0
    for _, entry in pairs(animals) do
        if type(entry) == "table" and entry.Placed == true then
            placed = placed + 1
            local ok, income = pcall(incomeOf, entry)
            if ok and Planner.isFinite(income) then weakest = math.min(weakest, income) end
        end
    end
    local records = {}
    for id, entry in pairs(animals) do
        if type(entry) ~= "table" then continue end
        local ok, income = pcall(incomeOf, entry)
        if not ok or not Planner.isFinite(income) then income = nil end
        local model = type(entry.ModelName) == "string" and entry.ModelName or "Unknown"
        local reason
        if eligible[id] then reason = "Weaker than every placed animal"
        elseif entry.Placed == true then reason = "On your plot"
        elseif entry.Bought then reason = "Purchased animal"
        elseif protectedNames[model:lower()] then reason = "Never-sell list"
        elseif entry.Favorite or entry.Favorited or entry.Locked or entry.Protected then reason = "Favorite / locked"
        elseif not validCapacity then reason = "Capacity unavailable"
        elseif placed < capacity then reason = "Empty plot slots"
        elseif not income or type(entry.ModelName) ~= "string" or entry.Placed ~= false then reason = "Data unavailable"
        elseif income >= weakest then reason = "Keep: equal or stronger"
        else reason = "Waiting for valid plot data" end
        records[#records + 1] = {Id = id, ModelName = model, Income = income,
            Eligible = eligible[id] == true, Reason = reason, Placed = entry.Placed == true}
    end
    table.sort(records, function(a, b)
        if a.Eligible ~= b.Eligible then return a.Eligible end
        if a.ModelName ~= b.ModelName then return a.ModelName < b.ModelName end
        return tostring(a.Id) < tostring(b.Id)
    end)
    return records
end
return Planner

local function numberText(n)
    if not Planner.isFinite(n) then return "—" end
    for _, unit in ipairs({{1e24,"Sp"},{1e21,"Sx"},{1e18,"Qi"},{1e15,"Qa"},{1e12,"T"},{1e9,"B"},{1e6,"M"},{1e3,"K"}}) do
        if math.abs(n) >= unit[1] then return string.format("%.2f%s", n / unit[1], unit[2]) end
    end
    return string.format("%.0f", n)
end
local function log(message)
    if not alive then return end
    table.insert(logEntries, 1, {Time = os.clock() - startedAt, Text = tostring(message)})
    if #logEntries > 10 then table.remove(logEntries) end
    for i, row in ipairs(logRows) do
        local e = logEntries[i]; row:Set(e and e.Text or "—")
        logTimes[i].Text = e and string.format("%02d:%02d", math.floor(e.Time / 60), math.floor(e.Time % 60)) or "—"
    end
end
local function failOnce(key, text)
    local now = os.clock()
    if not lastFailure[key] or now - lastFailure[key] > 20 then
        log(text); warn("[NoLimit] " .. text); lastFailure[key] = now
    end
end
local function get(key, fallback)
    if not ready then return fallback end
    local ok, value = pcall(modules.Data.Get, modules.Data, key)
    if ok and value ~= nil then return value end
    return fallback
end
local function blocked()
    local s = modules.State
    return s.Busy or s.OfflineOpen or s.TutorialActive or s.ShopOpen or s.ExclusiveShopOpen
        or s.IndexOpen or s.AnimalsOpen or s.SellOpen or s.InventoryOpen or s.ProtectedSellOpen
        or s.PackOfferShowing or s.ReleaseShow
end
local function setOff(key)
    opts[key] = false
    if switches[key] then switches[key]:Set(false, true) end
end
local function detach()
    for _, c in pairs({meterConnection, schedulerConnection, finishConnection, idleConnection}) do
        if c then c:Disconnect() end
    end
end
local function pause(value)
    if not alive then return end
    paused = value == true
    if pauseButton then pauseButton.Text = paused and "Resume all" or "Pause all" end
    if paused then stopTraining() end
    if alive then farmStatus:Set(paused and "Paused · finishing current action" or "Ready"); log(paused and "All automations paused" or "Automations resumed") end
end
local function invoke(name, ...)
    local remotes = ReplicatedStorage:FindFirstChild("Remotes")
    local remote = remotes and remotes:FindFirstChild(name)
    if not remote or not remote:IsA("RemoteFunction") then return {Ok = false, Reason = "Missing remote: " .. name} end
    local ok, result = pcall(remote.InvokeServer, remote, ...)
    if not ok then return {Ok = false, Reason = "Network error"} end
    if type(result) ~= "table" or type(result.Ok) ~= "boolean" then return {Ok = false, Reason = "Bad response"} end
    return result
end
local function perform(labelText, fn)
    if inFlight or not alive or paused then return end
    local token = {Name = labelText, Started = os.clock()}; inFlight = token
    worker = task.spawn(function()
        local key = ({Upgrade = "Upgrade", ["Equip food"] = "Upgrade", ["Equip best"] = "Equip",
            ["Sell animal"] = "Sell", Throw = "Farm", ["Offline earnings"] = "Offline"})[labelText]
        if not alive or paused or (checkGoals and checkGoals()) or (key and not opts[key]) then
            if inFlight == token then inFlight = nil end
            return
        end
        local ok, err = pcall(fn)
        if alive and not ok then failOnce(labelText, labelText .. " failed: " .. tostring(err)) end
        if inFlight == token then inFlight = nil end
        actionAfter = math.max(actionAfter, os.clock() + 2)
        if not alive and not ownThrow then detach() end
    end)
end
local function connectGame()
    if not alive or ready or loading then return end
    if moduleError then UI:Notify(moduleError); return end
    loading = true; loadingStarted = os.clock(); loadingWarned = false; connStatus:Set("Connecting…")
    task.spawn(function()
        local ok, err = pcall(function()
            assert(game.PlaceId == 120219252362013, "This script is for Throw Food For Animals.")
            local function load(parent, name)
                local obj = parent and parent:FindFirstChild(name)
                assert(obj and obj:IsA("ModuleScript"), "Missing module: " .. name)
                return require(obj) -- Reuse the game's modules. Never call their Init functions.
            end
            local scripts = player:FindFirstChild("PlayerScripts")
            local client = scripts and scripts:FindFirstChild("Client")
            local controllers = client and client:FindFirstChild("Controllers")
            assert(controllers, "Join Throw Food For Animals before enabling features.")
            local configs = ReplicatedStorage:FindFirstChild("Config")
            local shared = ReplicatedStorage:FindFirstChild("Shared")
            local nativeData = load(controllers, "DataController")
            assert(nativeData:IsReady(), "Player data is not ready in this module context. Rejoin and retry.")
            local staged = {Data = nativeData, State = load(controllers, "ClientState"),
                Throw = load(controllers, "ThrowController"), Train = load(controllers, "TrainingController"),
                Food = load(configs, "FoodsConfig"), Stand = load(configs, "StandsConfig"),
                Fence = load(configs, "FenceConfig"), Mutations = load(shared, "Mutations"),
                Offline = load(controllers, "OfflineEarnings")}
            assert(type(staged.Throw.Start) == "function" and type(staged.Train.Start) == "function"
                and type(staged.Train.Stop) == "function" and type(staged.Train.IsRunning) == "function"
                and type(staged.Offline.Claim) == "function" and type(staged.Offline.IsOpen) == "function", "Game controllers changed.")
            assert(type(staged.Mutations.IncomeOf) == "function", "Animal income calculation unavailable.")
            assert(staged.Data:IsReady(), "Player data is not ready in this module context. Rejoin and retry.")
            if not alive then return end
            modules = staged; ready = true
            for id in pairs(get("Animals", {})) do knownAnimals[id] = true end
            cashStart = cashStart or tonumber(get("Money", 0)) or 0
            connStatus:Set("Connected"); log("Connected to game data")
            if extras then extras.Refresh(true) end
            if modules.Throw.Finished and type(modules.Throw.Finished.Connect) == "function" then
                if finishConnection then finishConnection:Disconnect() end
                finishConnection = modules.Throw.Finished:Connect(function()
                    if alive then
                        stats.Throws = stats.Throws + 1
                        if scoreSent and scoreSent >= .92 then stats.Perfect = stats.Perfect + 1 end
                        if ownThrow then cycleThrows = cycleThrows + 1 end
                        log("Throw completed")
                    end
                    scoreSent = nil
                end)
            end
        end)
        loading = false
        if not ok and alive then
            moduleError = tostring(err); connStatus:Set("Not connected")
            for _, key in ipairs({"Farm", "Equip", "Upgrade", "Sell", "Perfect", "Offline"}) do setOff(key) end
            UI:Notify("Game connection failed. See Activity or F9 for details."); log(moduleError)
            warn("[NoLimit] " .. moduleError)
        end
    end)
end
local function toggle(section, key, title, desc)
    switches[key] = section:AddToggle({Id = key, Title = title, Description = desc, Default = opts[key],
        Callback = function(value)
            if value and (key == "Farm" or key == "Perfect" or key == "AntiAFK") and not inputService then
                UI:Notify("Automatic throw input is unavailable in this environment."); return false
            end
            if value and moduleError and key ~= "AntiAFK" and key ~= "Cycle" then UI:Notify("Reconnect the game first."); return false end
            opts[key] = value
            if key == "Sell" and not value then sellStatus:Set("Stopped") end
            if key == "Upgrade" and not value then upgradeStatus:Set("Stopped") end
            if key == "Offline" and not value and offlineStatus then offlineStatus:Set("Stopped") end
            if key == "Cycle" and not value then stopTraining() end
            if key == "Farm" or key == "Perfect" then inputFailed = false end
            if key == "Farm" then
                cycleThrows = 0; phase = "Training"
                if not value then stopTraining() end
            end
            if value and (key == "Farm" or key == "Perfect" or key == "Equip" or key == "Sell" or key == "Upgrade" or key == "Offline") then task.defer(connectGame) end
            log(title .. (value and " enabled" or " disabled"))
        end})
end

local function numberControl(section, key, title, desc, min, max, step)

        BackgroundColor3 = colors.bg, TextColor3 = colors.text, Font = Enum.Font.Gotham,

        if Planner.isFinite(value) then opts[key] = math.clamp(math.floor(value / step + .5) * step, min, max) end
        field.Text = tostring(opts[key])
    end)
    return field
end

local function tap(key)
    if not inputService then return false, "unavailable" end
    if UIS:GetFocusedTextBox() or UIS:IsKeyDown(key) then return false, "busy" end
    local down = pcall(function() inputService:SendKeyEvent(true, key, false, game) end)
    -- Always try to release the key, even when the press raised an error.
    local up = pcall(function() inputService:SendKeyEvent(false, key, false, game) end)
    return down and up, "unavailable"
end

meterConnection = RunService.RenderStepped:Connect(function()
    local seq = playerGui:FindFirstChild("ThrowSequence")
    -- Observe every sequence, even when assistance is off, so cancelled throws
    -- cannot donate a stale perfect-input score to a later manual throw.
    if seq and seq ~= observedSequence then
        observedSequence = seq; currentMarker = nil; scoreSent = nil
    end
    local shouldComplete = ownThrow
    if inputFailed or (not shouldComplete and (not alive or paused or not opts.Perfect)) then return end
    local meter = seq and seq:FindFirstChild("Meter")
    local bar = meter and meter:FindFirstChild("Bar")
    local marker = bar and bar:FindFirstChild("Marker")
    if not marker or not marker.Visible or not meter.Visible then currentMarker = nil; return end
    if marker ~= currentMarker then currentMarker = marker; sentAt = -math.huge; scoreSent = nil end
    local score = 1 - marker.Position.Y.Scale
    local target = opts.Perfect and .96 or .76
    if score >= target and os.clock() - sentAt > 1.6 then
        local ok, reason = tap(Enum.KeyCode.Space)
        if ok then scoreSent = score; sentAt = os.clock()
        elseif alive and reason ~= "busy" then
            inputFailed = true; inputFailedAt = os.clock(); setOff("Perfect"); setOff("Farm")
            UI:Notify("Automatic input failed. Complete the current throw manually.")
            log("Automatic input unavailable; stopped farming")
        end
    end
end)

local function runStep()
    if not ready or not alive then return end
    local list = observeData()
    if checkGoals() then return end
    if paused or cancelledNative then return end
    if inFlight then
        if os.clock() - inFlight.Started > (ownThrow and 120 or ownTraining and opts.TrainingSeconds + 30 or 20) and not timedOut then
            timedOut = true; pause(true); failOnce("timeout", "Action timed out; paused to avoid duplicate requests")
        end
        return
    end
    timedOut = false
    if opts.Offline and modules.Offline.IsOpen() and os.clock() >= offlineAfter then
        perform("Offline earnings", function()
            modules.Offline.Claim()
            offlineAfter = os.clock() + 10
            if alive then
                local done = not modules.Offline.IsOpen()
                offlineStatus:Set(done and "Claim window completed" or "Waiting before retry")
                if done then log("Offline earnings window completed") end
            end
        end)
        return
    end
    if ownTraining then
        if not modules.Train.IsRunning() then ownTraining = false
        elseif os.clock() >= trainDeadline or not opts.Farm then
            stopTraining()
        end
        return
    end
    if blocked() or UIS:GetFocusedTextBox() then farmStatus:Set("Waiting for game activity"); return end
    if os.clock() < actionAfter then return end
    local upgradeReady = true
    if pendingUpgrade then
        local matched
        if pendingUpgrade.Kind == "Food" then
            local owned = get("OwnedFoods", {})
            matched = owned[tostring(pendingUpgrade.Index)] == true or owned[pendingUpgrade.Index] == true
        else matched = (tonumber(get(pendingUpgrade.Kind == "Stand" and "Stand" or "FenceLevel", 0)) or 0) >= pendingUpgrade.Index end
        if matched then pendingUpgrade = nil
        elseif os.clock() - pendingUpgrade.Started > 12 then
            setOff("Upgrade"); upgradeStatus:Set("Stopped: waiting for game data")
            failOnce("pendingUpgrade", "Upgrade confirmation not replicated; stopped to prevent duplicate spending")
            upgradeReady = false
        else upgradeStatus:Set("Waiting for game data"); upgradeReady = false end
    end
    if opts.Upgrade and upgradeReady and os.clock() >= (backoff.Upgrade or 0) then
        local snapshot = {Money = cashNow, OwnedFoods = get("OwnedFoods", {}), Stand = get("Stand", 0), FenceLevel = get("FenceLevel", 0)}
        local item = Planner.nextUpgrade(snapshot, modules, priorityOrders[opts.Priority], opts, opts.Reserve)
        if item then
            perform("Upgrade", function()
                local remote = ({Food = "Shop_Buy", Stand = "Stand_Upgrade", Fence = "Plot_UpgradeFence"})[item.Kind]
                -- Recheck the reserve immediately before spending.
                if (tonumber(get("Money", 0)) or 0) - item.Price < opts.Reserve then return end
                local result = item.Kind == "Food" and invoke(remote, item.Index) or invoke(remote)
                if result.Ok then
                    pendingUpgrade = {Kind = item.Kind, Index = item.Index, Started = os.clock()}
                    stats.Upgrades = stats.Upgrades + 1; log(item.Kind .. " upgraded"); upgradeStatus:Set(item.Kind .. " upgraded")
                else
                    failOnce("upgrade", "Upgrade refused: " .. tostring(result.Reason))
                    backoff.Upgrade = os.clock() + 30; upgradeStatus:Set("Waiting before retry")
                end
            end); return
        end
        upgradeStatus:Set("Waiting for funds / maximum reached")
        if opts.Food then
            local owned, highest = get("OwnedFoods", {}), 0
            for i = 1, modules.Food.Count() do if owned[tostring(i)] or owned[i] then highest = i end end
            if highest > 0 and get("EquippedFood", 0) ~= highest and os.clock() >= (backoff.FoodEquip or 0) then
                perform("Equip food", function()
                    local r = invoke("Shop_Equip", highest)
                    if r.Ok then log("Best owned food equipped") else failOnce("food", "Food equip refused: " .. tostring(r.Reason)) end
                    backoff.FoodEquip = os.clock() + 30
                end); return
            end
        end
    end
    if opts.Equip and os.clock() >= (backoff.Equip or 0) then
        local fp = inventoryFingerprint(list)
        if fp ~= lastEquip then
            perform("Equip best", function()
                local r = invoke("Plot_EquipBest")
                if r.Ok then lastEquip = inventoryFingerprint(get("Animals", {})); log("Best animals equipped")
                else failOnce("equip", "Equip Best refused: " .. tostring(r.Reason)) end
                backoff.Equip = os.clock() + (r.Ok and 3 or 30)
            end); return
        end
    end
    if opts.Sell then
        local candidates = Planner.sellCandidates(list, plotCapacity(), animalIncome, protectedNames)
        for i = #candidates, 1, -1 do if soldIds[candidates[i].Id] then table.remove(candidates, i) end end
        local candidate = candidates[1]
        sellStatus:Set(candidate and ("Eligible: " .. #candidates) or "No eligible inventory animals")
        if candidate then
            perform("Sell animal", function()
                -- Re-read all protection and placement conditions immediately before sale.
                local fresh = Planner.sellCandidates(get("Animals", {}), plotCapacity(), animalIncome, protectedNames)
                local stillEligible = false
                for _, entry in ipairs(fresh) do if entry.Id == candidate.Id and not soldIds[entry.Id] then stillEligible = true; break end end
                if not stillEligible then return end
                local r = invoke("Plot_SellAnimal", candidate.Id)
                if r.Ok then
                    soldIds[candidate.Id] = true
                    stats.Sales = stats.Sales + 1; stats.SaleMoney = stats.SaleMoney + (tonumber(r.Money) or 0)
                    log("Sold " .. tostring(candidate.Entry.ModelName) .. " · " .. numberText(tonumber(r.Money) or 0))
                else setOff("Sell"); sellStatus:Set("Stopped: " .. tostring(r.Reason)); log("Auto Sell stopped: " .. tostring(r.Reason)) end
            end); return
        end
    end
    if opts.Farm and os.clock() >= (backoff.Farm or 0) then
        if opts.Cycle and (phase == "Training" or cycleThrows >= opts.ThrowsPerCycle) and (tonumber(get("Stand", 0)) or 0) > 0 then
            phase = "Throws"; cycleThrows = 0; ownTraining = true; trainStopAsked = false; stopRequestedAt = nil
            trainDeadline = os.clock() + opts.TrainingSeconds
            trainStatus:Set("Training for " .. opts.TrainingSeconds .. "s")
            -- Training.Start yields for the duration. It must not occupy the action lock,
            -- because the scheduler needs to call Stop when the deadline expires.
            task.spawn(function()
                if not alive or paused or checkGoals() or not opts.Farm or not opts.Cycle then ownTraining = false; return end
                trainingThread = coroutine.running(); trainingStarted = os.clock()
                local ok, err = pcall(modules.Train.Start)
                ownTraining = false; trainingThread = nil; trainingStarted = nil
                if alive then
                    if not ok then failOnce("training", "Training failed: " .. tostring(err)) end
                    trainStatus:Set("Throw phase"); actionAfter = os.clock() + 2
                else detach() end
            end)
            return
        end
        perform("Throw", function()
            throwThread = coroutine.running(); throwStarted = os.clock(); inputFailedAt = nil; scoreSent = nil
            ownThrow = true; farmStatus:Set("Throwing · waiting for normal finish")
            local before = stats.Throws
            local ok, err = pcall(modules.Throw.Start)
            ownThrow = false; throwThread = nil; throwStarted = nil
            if alive then
                if not ok then setOff("Farm"); log("Auto Farm stopped: " .. tostring(err))
                elseif stats.Throws == before then
                    farmFailures = farmFailures + 1
                    backoff.Farm = os.clock() + (farmFailures >= 3 and 15 or 3)
                    failOnce("throw", "Throw did not complete; waiting before retry")
                else farmFailures = 0 end
                farmStatus:Set(opts.Farm and "Ready for next throw" or "Stopped")
            end
        end)
    else farmStatus:Set("Stopped") end
end
local function supervise()
    local now = os.clock()
    if alive and loading and now - loadingStarted > 20 and not loadingWarned then
        loadingWarned = true; connStatus:Set("Loading is taking longer")
        log("Game module loading has not returned. No duplicate connection is started.")
        UI:Notify("Game loading is taking longer. See Activity for details.")
    end
    local expiredThrow = ownThrow and throwThread and ((throwStarted and now - throwStarted > 120)
        or (closingAt and now - closingAt > 12) or (inputFailedAt and now - inputFailedAt > 8))
    local expiredTrain = ownTraining and trainingThread and ((stopRequestedAt and now - stopRequestedAt > 8)
        or (closingAt and now - closingAt > 12) or (trainingStarted and now - trainingStarted > opts.TrainingSeconds + 30))
    if expiredThrow or expiredTrain then
        local kind = expiredThrow and "Throw" or "Training"
        local thread = expiredThrow and throwThread or trainingThread
        -- Cancel only our own suspended Start coroutine. The supplied game
        -- controllers' checkDeadThrow/checkDeadTraining then run their cleanup.
        local ok = pcall(task.cancel, thread)
        if ok then
            if expiredThrow then ownThrow = false; throwThread = nil; inFlight = nil; scoreSent = nil
            else ownTraining = false; trainingThread = nil end
            cancelledNative = {Kind = kind, Time = now}
            if alive then
                setOff("Farm"); pause(true); log("Stopped a stuck " .. kind:lower() .. "; waiting for game cleanup")
            end
        end
    end
    if cancelledNative then
        if modules.State.Busy ~= cancelledNative.Kind then
            cancelledNative = nil; actionAfter = now + 2
        elseif now - cancelledNative.Time > 5 then
            if alive then
                setOff("Farm"); log("Game cleanup is still pending. Rejoin before restarting.")
                UI:Notify("Game cleanup did not finish. Rejoin before restarting.")
            end
            cancelledNative.Time = math.huge
            if not alive then detach() end
        end
    end
    if not alive and not ownThrow and not ownTraining and not cancelledNative then detach() end
end
local tickTime = 0
schedulerConnection = RunService.Heartbeat:Connect(function(dt)
    tickTime = tickTime + dt
    if tickTime < .5 then return end
    tickTime = 0
    if ownTraining and (paused or not opts.Farm or not opts.Cycle or os.clock() >= trainDeadline) then stopTraining() end
    supervise()
    if not alive then return end
    local ok, err = pcall(runStep)
    if not ok and alive then pause(true); failOnce("scheduler", "Paused after an error: " .. tostring(err)) end
end)

    alive = false; closingAt = os.clock()
    if idleConnection then idleConnection:Disconnect() end
    stopTraining()
    -- The regular throw owns camera/character cleanup. Let this one finish; keep only
    -- its meter assistance until it returns. No new actions can be scheduled.
    if not ownThrow and not ownTraining then detach() end
end)
log("All automations start switched off")
end
